cmake_minimum_required(VERSION 3.27)
project(XVFS LANGUAGES CXX C)

# Set architecture (x64)
set(CMAKE_GENERATOR_PLATFORM x64)

# Set build configurations
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Debug CACHE STRING "Choose the type of build." FORCE)
    set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS "Debug" "Release")
endif()

# Output directories
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/build/${CMAKE_BUILD_TYPE})
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/build/${CMAKE_BUILD_TYPE})
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/build/${CMAKE_BUILD_TYPE})

# Source files
file(GLOB_RECURSE XVFS_SOURCES
    "src/*.c"
    "src/*.cpp"
    "src/*.h"
    "src/*.hpp"
)

# Define static library
add_library(XVFS STATIC ${XVFS_SOURCES})

# Include directories
target_include_directories(XVFS PUBLIC include)

# C++ standard
set_target_properties(XVFS PROPERTIES
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED YES
)
